    /// <summary>Agent, ordinary operator, inspected operator and quarantine administrator keep the original principal and distinct grants.</summary>
    [Theory]
    [InlineData(true, false, false, "agent", "ProjectUser")]
    [InlineData(false, false, false, "ui", "TenantOperator")]
    [InlineData(false, true, false, "ui", "TenantOperator")]
    [InlineData(false, true, true, "ui", "TenantProjectAdministrator")]
    [InlineData(false, false, true, "ui", "TenantProjectAdministrator")]
    public void NativeUserSurfacesPreservePrincipalAndInspectionPermission(bool chatbot, bool inspect, bool projectAdministrator,
        string surface, string role)
    {
        List<Claim> claims = PackageReadFixture.HumanClaims("primary", "foreign", chatbot, inspect, projectAdministrator: projectAdministrator);
        claims.Single(claim => claim.Type == "sub").Value.ShouldBe("actor-1");
        claims.Single(claim => claim.Type == "eventstore:surface").Value.ShouldBe(surface);
        claims.Single(claim => claim.Type == "azp").Value.ShouldBe(chatbot ? "chatbot" : "operator-web");
        claims.Single(claim => claim.Type == "projects:role").Value.ShouldBe(role);
        claims.Any(claim => claim.Type == "eventstore:permission" && claim.Value == "projects:inspect_descriptive_metadata").ShouldBe(inspect);
        claims.ShouldNotContain(claim => claim.Type == "global_admin" || claim.Type == "is_global_admin");
        claims.Where(claim => claim.Type == "eventstore:tenant").Select(claim => claim.Value).ShouldBe(["primary"]);
    }

    /// <summary>Foreign denial reaches coarse transport admission as the original primary actor with no foreign membership claim.</summary>
    [Fact]
    public void ForeignCoarseGrantDoesNotChangeActorOrPublicTenantIdentity()
    {
        List<Claim> claims = PackageReadFixture.HumanClaims("primary", "foreign", foreignTransportGrant: true);
        claims.Where(claim => claim.Type == "eventstore:tenant").Select(claim => claim.Value).ShouldBe(["primary", "foreign"]);
        claims.Single(claim => claim.Type == "sub").Value.ShouldBe("actor-1");
        claims.Single(claim => claim.Type == "tenant_id").Value.ShouldBe("primary");
        claims.ShouldNotContain(claim => claim.Value == "system" || claim.Value == "foreign-actor");
        List<Claim> administrator = PackageReadFixture.HumanClaims("primary", "foreign", chatbot: false, administrator: true);
        administrator.ShouldContain(claim => claim.Type == "global_admin" && claim.Value == "true");
        administrator.ShouldContain(claim => claim.Type == "eventstore:permission" && claim.Value == "tenants:create");
        administrator.Where(claim => claim.Type == "eventstore:tenant").Select(claim => claim.Value).ShouldBe(["primary"]);
    }
