    private static async Task VerifyTelemetryAsync(PackageApplicationFactory factory, DistributedApplication application,
        FixtureScratch scratch, CancellationToken cancellation)
    {
        // The login URL contains a credential; keep it private and retain only endpoint authorities.
        Uri dashboard = await application.GetDashboardUrlAsync(cancellation).ConfigureAwait(false);
        Uri[] endpoints = [dashboard, factory.GetEndpoint("aspire-dashboard", "otlp-grpc"), factory.GetEndpoint("aspire-dashboard", "otlp-http")];
        foreach (Uri endpoint in endpoints)
        {
            if (!endpoint.IsLoopback || endpoint.Port <= 0 || endpoint.UserInfo.Length != 0)
                throw new InvalidOperationException("Package telemetry requires its owned allocated loopback receiver.");
            using System.Net.Sockets.TcpClient connection = new();
            await connection.ConnectAsync(endpoint.Host, endpoint.Port, cancellation).ConfigureAwait(false);
        }
        var configuration = application.Services.GetRequiredService<Microsoft.Extensions.Configuration.IConfiguration>();
        string key = configuration["AppHost:OtlpApiKey"] ?? throw new InvalidOperationException("Private telemetry authentication is missing.");
        using HttpClient client = new(new HttpClientHandler { AllowAutoRedirect = false }) { Timeout = TimeSpan.FromSeconds(10) };
        async Task<System.Net.HttpStatusCode> ExportAsync(bool authenticate)
        {
            using HttpRequestMessage request = new(HttpMethod.Post, new Uri(endpoints[2], "/v1/traces"));
            request.Content = new ByteArrayContent([]);
            request.Content.Headers.ContentType = new System.Net.Http.Headers.MediaTypeHeaderValue("application/x-protobuf");
            if (authenticate) request.Headers.Add("x-otlp-api-key", key);
            using HttpResponseMessage response = await client.SendAsync(request, cancellation).ConfigureAwait(false);
            return response.StatusCode;
        }
        if (await ExportAsync(false).ConfigureAwait(false) is not (System.Net.HttpStatusCode.Unauthorized or System.Net.HttpStatusCode.Forbidden)
            || await ExportAsync(true).ConfigureAwait(false) != System.Net.HttpStatusCode.OK)
            throw new InvalidOperationException("The actual telemetry receiver did not enforce its private export key.");
        using GrpcChannel channel = GrpcChannel.ForAddress(endpoints[1]);
        Marshaller<byte[]> bytes = Marshallers.Create<byte[]>(value => value, value => value);
        Method<byte[], byte[]> export = new(MethodType.Unary, "opentelemetry.proto.collector.trace.v1.TraceService", "Export", bytes, bytes);
        async Task<bool> ExportGrpcAsync(bool authenticate)
        {
            Metadata headers = [];
            if (authenticate) headers.Add("x-otlp-api-key", key);
            using AsyncUnaryCall<byte[]> call = channel.CreateCallInvoker().AsyncUnaryCall(export, host: null,
                new CallOptions(headers, DateTime.UtcNow.AddSeconds(10), cancellation), []);
            try { _ = await call.ResponseAsync.ConfigureAwait(false); return true; }
            catch (RpcException error) when (error.StatusCode is StatusCode.Unauthenticated or StatusCode.PermissionDenied) { return false; }
        }
        if (await ExportGrpcAsync(false).ConfigureAwait(false) || !await ExportGrpcAsync(true).ConfigureAwait(false))
            throw new InvalidOperationException("The actual gRPC telemetry receiver did not enforce its private export key.");
        using JsonDocument process = JsonDocument.Parse(await File.ReadAllBytesAsync(scratch.File("projects-process.json"), cancellation).ConfigureAwait(false));
        JsonElement workload = process.RootElement;
        int selectedPort = workload.GetProperty("otlpProtocol").GetString() switch
        {
            "grpc" => endpoints[1].Port, "http/protobuf" => endpoints[2].Port, _ => -1,
        };
        string expectedKeyHash = Convert.ToHexStringLower(SHA256.HashData(System.Text.Encoding.UTF8.GetBytes(key)));
        if (!workload.GetProperty("otlpLoopback").GetBoolean() || workload.GetProperty("otlpPort").GetInt32() != selectedPort
            || workload.GetProperty("otlpApiKeySha256").GetString() != expectedKeyHash)
            throw new InvalidOperationException("The actual Projects exporter is not bound to its authenticated allocated telemetry receiver.");
        await File.WriteAllTextAsync(scratch.File("telemetry-receiver.json"), JsonSerializer.Serialize(new
        {
            ownedLoopbackEndpointsVerified = true, unauthenticatedExportRefused = true, authenticatedExportAccepted = true,
            unauthenticatedGrpcExportRefused = true, authenticatedGrpcExportAccepted = true, actualWorkloadExporterVerified = true,
            ports = endpoints.Select(endpoint => endpoint.Port).ToArray(),
        }), cancellation).ConfigureAwait(false);
    }

    private static async Task BrowserAsync(string phase, string[] selectors, string[] extra, string workspace,
        Dictionary<string, string> environment, FixtureInputs inputs, FixtureScratch scratch, CancellationToken cancellationToken)
    {
        environment["E2E_RUN_ID"] = phase == "actual-owner" ? scratch.Isolation : scratch.Isolation + "-" + phase;
        environment["E2E_FIXTURE_RETRY_PROBE"] = phase == "retry" ? "1" : "0";
        environment["E2E_FIXTURE_RETRY_DIRECTORY"] = scratch.File("retry-probes");
        environment["E2E_PROPOSAL_OBSERVATIONS_FILE"] = scratch.File(phase + "-proposal-observations.jsonl");
        environment["E2E_PACKAGE_OUTPUT"] = scratch.File(phase + "-output");
        string[] arguments = [Path.Combine(workspace, "node_modules/@playwright/test/cli.js"), "test", .. selectors,
            "--project", "chromium", "--workers", "2", "--retries", "0", .. extra];
        string output = await FixtureProcess.RunAsync("node", arguments, workspace, cancellationToken,
            environment, inputs.ProtectedValues, printOutput: true).ConfigureAwait(false);
        Match counts = Regex.Match(output, @"\[zero-live-skip\] collected=(\d+) skipped=(\d+) status=passed");
        if (!counts.Success || int.Parse(counts.Groups[1].Value, System.Globalization.CultureInfo.InvariantCulture) <= 0
            || counts.Groups[2].Value != "0")
            throw new InvalidOperationException("The complete browser phase did not prove a nonempty zero-skip result: " + phase);
    }

    private static async Task VerifyReadRecoveryAsync(string root, PackageApplicationFactory factory, FixtureInputs inputs,
        FixtureScratch scratch, CancellationToken cancellationToken)
    {
        const string method = "Hexalith.Projects.Server.Tests.Queries.SupportedProjectReadHttpTests.ProtectedSdkDispatchAndReadsSurviveHostAndStoreClientRestart";
        string assembly = Path.Combine(root, "tests/Hexalith.Projects.Server.Tests/bin/Release/net11.0/Hexalith.Projects.Server.Tests.dll");
        string result = scratch.File("read-recovery.json");
        Dictionary<string, string> environment = new()
        {
            ["STORY61_DAPR_HTTP_ENDPOINT"] = factory.GetEndpoint("projects-dapr-cli", "http").ToString(),
            ["STORY61_DAPR_GRPC_ENDPOINT"] = factory.GetEndpoint("projects-dapr-cli", "grpc").ToString(),
        };
        await FixtureProcess.RunAsync("dotnet", [assembly, "-method", method, "-noColor", "-result-ctrf", result],
            root, cancellationToken, environment, inputs.ProtectedValues, printOutput: true).ConfigureAwait(false);
        using JsonDocument report = JsonDocument.Parse(await File.ReadAllBytesAsync(result, cancellationToken).ConfigureAwait(false));
        JsonElement tests = report.RootElement.GetProperty("results").GetProperty("tests");
        if (tests.GetArrayLength() != 1 || tests[0].GetProperty("status").GetString() != "passed")
            throw new InvalidOperationException("The real SDK host/store-client restart assertion did not pass without skips.");
    }
