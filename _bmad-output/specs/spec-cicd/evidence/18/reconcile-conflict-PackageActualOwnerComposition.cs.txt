    /// <summary>Returns whether the profile was explicitly selected.</summary>
    public static bool IsEnabled(IConfiguration configuration)
    {
        ArgumentNullException.ThrowIfNull(configuration);
        return string.Equals(configuration[EnabledConfigurationKey], "true", StringComparison.OrdinalIgnoreCase)
            || string.Equals(configuration[EnabledConfigurationKey], "1", StringComparison.Ordinal);
    }

    /// <summary>Loads and validates launcher-owned inputs before normal resources are composed.</summary>
    public static void Configure(IDistributedApplicationBuilder builder)
    {
        ArgumentNullException.ThrowIfNull(builder);
        if (!IsEnabled(builder.Configuration)) return;

        if (!builder.Environment.IsDevelopment()) throw new InvalidOperationException("Supported-read fixtures require Development.");
        if (string.Equals(builder.Configuration["EnableKeycloak"], "true", StringComparison.OrdinalIgnoreCase)
            || string.Equals(builder.Configuration["KeycloakPersistent"], "true", StringComparison.OrdinalIgnoreCase))
            throw new InvalidOperationException("Conflicting fixture or persistent security configuration cannot be overridden by fixture inputs.");
        string path = Path.GetFullPath(Required(builder.Configuration, ConfigurationFileKey));
        if (!File.Exists(path)) throw new InvalidOperationException("The private fixture configuration is missing.");
        if (!OperatingSystem.IsWindows()
            && (File.GetUnixFileMode(path) & (UnixFileMode.GroupRead | UnixFileMode.GroupWrite | UnixFileMode.OtherRead | UnixFileMode.OtherWrite)) != 0)
            throw new InvalidOperationException("Fixture configuration must be private to its owner.");
        builder.Configuration.AddJsonFile(path, optional: false, reloadOnChange: false);
        Validate(builder.Configuration);
    }

                .WithEnvironment(prefix + "NotBefore", DateTimeOffset.UtcNow.AddMinutes(-1).ToString("O"))
                .WithEnvironment(prefix + "ExpiresAt", DateTimeOffset.UtcNow.AddHours(2).ToString("O"));
            for (int i = 0; i < operations.Length; i++) _ = authority.WithEnvironment(prefix + $"Operations__{i}", operations[i]);
            if (queryBindings)
            {
                _ = authority.WithEnvironment(prefix + "Bindings__eventstore:bound-tenant__0", Required(configuration, Prefix + "Tenant"))
                    .WithEnvironment(prefix + "Bindings__eventstore:bound-tenant__1", Required(configuration, Prefix + "ForeignTenant"))
                    .WithEnvironment(prefix + "Bindings__eventstore:bound-domain__0", audience);
            }
        }
    }

    private static string AddQualificationResiliency(IConfiguration configuration, string resourcesPath, string appId)
    {
        if (appId is not ("eventstore" or "projects")) throw new InvalidOperationException("Only the two actual audit-writing sidecars select this qualification policy.");
        string directory = Path.Combine(Path.GetDirectoryName(Required(configuration, Prefix + "CertificatePath"))!,
            appId == "eventstore" ? "components" : "components-projects");
        _ = Directory.CreateDirectory(directory);
        string path = Path.Combine(directory, appId == "eventstore" ? "owner-resiliency.yaml" : "projects-resiliency.yaml");
        object baseline = new DeserializerBuilder().WithAttemptingUnquotedStringTypeDeserialization().Build()
            .Deserialize<object>(File.ReadAllText(Path.Combine(resourcesPath, "resiliency.yaml")));
        JsonObject policy = JsonNode.Parse(new SerializerBuilder().JsonCompatible().Build().Serialize(baseline))!.AsObject();
        if (policy["apiVersion"]?.GetValue<string>() != "dapr.io/v1alpha1" || policy["kind"]?.GetValue<string>() != "Resiliency")
            throw new InvalidOperationException("The actual baseline resiliency policy is required.");
        JsonObject policies = policy["spec"]!["policies"]!.AsObject();
        JsonObject apps = policy["spec"]!["targets"]!["apps"]!.AsObject();
        JsonObject projects = apps["projects"]!.AsObject();
        if (apps.ContainsKey("folders") || projects["timeout"] is null
            || policies["timeouts"]?[projects["timeout"]!.GetValue<string>()] is null)
            throw new InvalidOperationException("The private profile requires the existing Project timeout and no competing Folder target.");
        policy["metadata"]!["name"] = appId == "eventstore" ? "supported-reads-owner-deadline" : "supported-reads-project-state";
        policy["scopes"] = new JsonArray(JsonValue.Create(appId));
        policies["retries"]!["supportedReadsOwnerNoRetry"] = new JsonObject
        {
            ["policy"] = "constant", ["duration"] = "1ms", ["maxRetries"] = 0,
        };
        JsonObject state = policy["spec"]!["targets"]!["components"]!["statestore"]!["outbound"]!.AsObject();
        if (state["timeout"] is null || policies["timeouts"]?[state["timeout"]!.GetValue<string>()]?.GetValue<string>() != "15s")
            throw new InvalidOperationException("Qualification must retain the actual fifteen-second state timeout.");
        state["retry"] = "supportedReadsOwnerNoRetry";
        _ = state.Remove("circuitBreaker");
        if (appId == "eventstore")
        {
            policies["timeouts"]!["supportedReadsOwnerTimeout"] = "2s";
            projects["retry"] = "supportedReadsOwnerNoRetry";
            _ = projects.Remove("circuitBreaker");
            apps["folders"] = new JsonObject { ["timeout"] = "supportedReadsOwnerTimeout", ["retry"] = "supportedReadsOwnerNoRetry" };
        }
        File.WriteAllText(path, policy.ToJsonString());
        if (!OperatingSystem.IsWindows()) File.SetUnixFileMode(path, UnixFileMode.UserRead | UnixFileMode.UserWrite);
        return directory;
    }

    private static void ValidateFolderSourceKeys(IConfiguration configuration)
    {
        HashSet<string> fingerprints = new(StringComparer.Ordinal);
        foreach (string prefix in new[] { "FolderOwner", "FolderScan", "FolderWorker", "FolderWorkerScan" })
        {
            try
            {
                byte[] privateBytes = Convert.FromBase64String(Required(configuration, Prefix + prefix + "PrivateKey"));
                byte[] publicBytes = Convert.FromBase64String(Required(configuration, Prefix + prefix + "PublicKey"));
                using RSA privateKey = RSA.Create();
                using RSA publicKey = RSA.Create();
                privateKey.ImportPkcs8PrivateKey(privateBytes, out int privateRead);
                publicKey.ImportSubjectPublicKeyInfo(publicBytes, out int publicRead);
                byte[] actualPublic = privateKey.ExportSubjectPublicKeyInfo();
                if (privateRead != privateBytes.Length || publicRead != publicBytes.Length || privateKey.KeySize < 2048
                    || !actualPublic.AsSpan().SequenceEqual(publicKey.ExportSubjectPublicKeyInfo())
                    || !fingerprints.Add(Convert.ToHexString(SHA256.HashData(actualPublic))))
                    throw new InvalidOperationException("Folder source read/scan proof keys must match and remain independent: " + prefix);
            }
            catch (Exception error) when (error is FormatException or CryptographicException)
            {
                throw new InvalidOperationException("Invalid Folder source proof key material: " + prefix, error);
            }
        }
        if (Required(configuration, Prefix + "FoldersSecret") == Required(configuration, Prefix + "FolderWorkersSecret"))
            throw new InvalidOperationException("Folder source workloads require independent issuer secrets.");
    }

    private static string Required(IConfiguration configuration, string key)
        => !string.IsNullOrWhiteSpace(configuration[key]) ? configuration[key]! : throw new InvalidOperationException("Missing supported-read fixture configuration: " + key);

    private static string ResolveProjectRoot(string directory)
    {
        foreach (string start in new[] { directory, Directory.GetCurrentDirectory() })
            for (DirectoryInfo? candidate = new(start); candidate is not null; candidate = candidate.Parent)
                if (File.Exists(Path.Combine(candidate.FullName, "Hexalith.Projects.slnx"))) return candidate.FullName;
        throw new DirectoryNotFoundException("The supported-read profile requires the owning Projects checkout.");
    }
